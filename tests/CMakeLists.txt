cmake_minimum_required(VERSION 3.24)
project(text_resource_tests LANGUAGES C CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Use MSVC and -A Win32: the production hook uses x86 thiscall/assembly")
endif()
include(FetchContent)
FetchContent_Declare(lua URL https://www.lua.org/ftp/lua-5.4.6.tar.gz
  URL_HASH SHA256=7d5ea1b9cb6aa0b59ca3dde1c6adcb57ef83a1ba8e5432c0ecd06bf439b3ad88)
FetchContent_MakeAvailable(lua)
file(GLOB lua_sources "${lua_SOURCE_DIR}/src/*.c")
list(REMOVE_ITEM lua_sources "${lua_SOURCE_DIR}/src/lua.c" "${lua_SOURCE_DIR}/src/luac.c")
add_library(lua STATIC ${lua_sources})
target_include_directories(lua PUBLIC "${lua_SOURCE_DIR}/src")
set(module_dir "${CMAKE_CURRENT_SOURCE_DIR}/../ucp_textResourceModifier/ucp_textResourceModifier")
add_executable(text_resource_tests text_resource_tests.cpp "${module_dir}/textResourceModifier.cpp")
target_compile_features(text_resource_tests PRIVATE cxx_std_17)
target_compile_options(text_resource_tests PRIVATE /utf-8 /Zc:threadSafeInit-)
target_include_directories(text_resource_tests PRIVATE stubs "${module_dir}")
target_link_libraries(text_resource_tests PRIVATE lua)
enable_testing()
add_test(NAME composition COMMAND text_resource_tests composition)
add_test(NAME reload COMMAND text_resource_tests reload)
